# Profile-scoped variants of canonical pages. These are duplicates of the
# canonical pages on the <username> axis and dilute crawl budget. Canonical
# /games/<np_id>/ and /my-pursuit/badges/<badge>/ pages remain crawlable.
Disallow: /games/*/*
Disallow: /my-pursuit/badges/*/*

# Filtered / paginated / tabbed variants of a profile. The canonical
# /community/profiles/<username>/ page stays crawlable -- profiles are the free
# floor of the product and should keep their search and share value -- but the
# ?tab= / ?page= / ?sort= permutations are duplicates of it on axes that multiply
# out into an unbounded crawl space, and each one costs a real query.
#
# The `/` before `?` is load-bearing: `*` matches the empty string, so the simpler
# /community/profiles/*?* would also match /community/profiles/?page=2 and block the
# profile INDEX's pagination -- the very thing a crawler should walk to discover
# profiles. Requiring a path segment first confines this to profile detail URLs.
Disallow: /community/profiles/*/?*
